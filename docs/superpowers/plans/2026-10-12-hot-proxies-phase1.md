# Hot Proxies Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Chrome PWA that turns a typed card name into a 384 × 536 dot black-and-white proxy image for a 58 mm thermal printer, previewed in the app and downloadable as PNG.

**Architecture:** A framework-free TypeScript core (`src/core`) renders a card from a normalized `CardData` onto a Canvas 2D context through an injected `RenderEnv`, so the same code runs in the browser and in Node tests via `@napi-rs/canvas`. A thin data layer provides a local fuzzy name index and a rate-limited Scryfall client with an injected cache. A React + MUI shell wires search, preview and settings together.

**Tech Stack:** Vite 8, React 19, TypeScript 5, MUI 9, vite-plugin-pwa 2, idb-keyval 6, Vitest 5, @napi-rs/canvas 1, @testing-library/react 16, jsdom, tsx. Node 24.

**Spec:** `docs/superpowers/specs/2026-10-12-hot-proxies-design.md`

## Global Constraints

- Card grid: `DOTS_PER_LINE = 384`, `DPI = 203`, height = `Math.round(width * 88 / 63)` = 536. Both live in `src/core/printer.ts` and are overridable through settings.
- Only `layout === 'normal'` cards render; anything else throws `UnsupportedLayout`.
- Proxy content is exactly: name, mana cost, type line, rules text, P/T or loyalty. No set code, artist, flavor text or color indicator.
- Default settings: dither `atkinson`, block `1`, gamma `1.2`, contrast cutoff `1`, text size `17`, name size `22`, strip reminder `true`, dots per line `384`, dpi `203`, font family `ProxySerif`.
- Fit order (spec 5.5): art 250→180 in steps of 10 at base text size; then text size base→14 at art 180; then art 170→100 at text 14; then strip reminder text (if not already) and repeat from the text-size step; then text 13→11; then clip and flag `overflow`.
- Text is thresholded at 50 %, never dithered. Art is dithered, never thresholded through the text path.
- Fonts: Crimson Text Regular and Bold TTF, registered under the single family name `ProxySerif` in both browser and Node.
- Mana symbols come only from `src/assets/mana/paths.ts`, generated from `mana-master/svg` by `scripts/vendor-mana.ts`. Unknown symbols render as literal text in a circle and never throw.
- TypeScript `strict: true`. No `any` outside the two `RenderEnv` adapters.
- All commands run from the repo root `/home/aure/github/hot-proxies`.
- The sandbox makes `~/.npm/_cacache` read-only. Prefix every npm command with `npm_config_cache="$TMPDIR/npm-cache"` (or export it once per shell). If `npm install` still fails with `EROFS`, rerun it outside the sandbox.
- Each task ends with a commit. Commit messages end with the attribution lines in effect for this session.

## Review Focus

Inputs the spec implies but names no test for, most likely to bite first. Each one has a pinned test in the task named.

1. **A single word wider than the text box** (e.g. "Asmoranomardicadaistinaculdacar" at 17 px in a 348-dot box) must be broken by characters, not loop forever or overflow. Test in Task 7 (`wrap.test.ts`, "breaks an overlong word by characters").
2. **A paragraph that is only reminder text** (e.g. a basic land's "({T}: Add {G}.)") must vanish entirely when stripped, leaving no empty line. Test in Task 9 (`reminder.test.ts`, "removes a paragraph that is only reminder text").
3. **Names with non-ASCII letters** ("Lim-Dûl's Vault", "Æther Vial") must be found by typing plain ASCII and must be URL-encoded in the Scryfall request. Tests in Task 12 (`fuzzy.test.ts`, "matches ignoring diacritics") and Task 13 (`scryfall.test.ts`, "URL-encodes the name").
4. **Portrait or tiny art** (art crop narrower than tall, or smaller than the art box) must still fill the box edge to edge by cover-resizing, never letterbox or stretch. Test in Task 8 (`resize.test.ts`, "cover-resizes a portrait source").
5. **Corrupt settings in localStorage** (invalid JSON, wrong types, out-of-range numbers) must fall back to defaults field by field, never crash startup. Test in Task 2 (`settings.test.ts`, "clamps and defaults a corrupt object").

```ts
/** PT-210 class printer: 58 mm roll, 48 mm printable, 203 dpi. */
export const DOTS_PER_LINE = 384
export const DPI = 203
/** Card height in dots for a given width, keeping the real 63:88 card ratio. */
export function cardHeight(width: number): number {
  return Math.round((width * 88) / 63)
}
```

```ts
import { DOTS_PER_LINE, DPI } from './printer'

export type DitherName = 'atkinson' | 'floyd-steinberg' | 'bayer4' | 'threshold'
export const DITHER_NAMES: DitherName[] = ['atkinson', 'floyd-steinberg', 'bayer4', 'threshold']
export type FontFamilyName = 'ProxySerif' | 'MPlantin'

export interface Settings {
  dither: DitherName
  block: 1 | 2
  gamma: number          // 0.8 .. 1.6
  contrastCutoff: number // 0 .. 5 (percent)
  textSize: number       // 14 .. 22
  nameSize: number       // 16 .. 26
  stripReminder: boolean
  dotsPerLine: number    // 192 .. 832
  dpi: number            // 100 .. 400
  fontFamily: FontFamilyName
}

export const DEFAULT_SETTINGS: Settings = {
  dither: 'atkinson', block: 1, gamma: 1.2, contrastCutoff: 1,
  textSize: 17, nameSize: 22, stripReminder: true,
  dotsPerLine: DOTS_PER_LINE, dpi: DPI, fontFamily: 'ProxySerif',
}

export const RANGES = {
  gamma: [0.8, 1.6], contrastCutoff: [0, 5], textSize: [14, 22], nameSize: [16, 26],
  dotsPerLine: [192, 832], dpi: [100, 400],
} as const

function num(v: unknown, [lo, hi]: readonly [number, number], dflt: number): number {
  if (typeof v !== 'number' || !Number.isFinite(v)) return dflt
  return Math.min(hi, Math.max(lo, v))
}

/** Coerce anything (e.g. parsed localStorage) into a valid Settings, field by field. */
export function normalizeSettings(input: unknown): Settings {
  const o = (input && typeof input === 'object' ? input : {}) as Record<string, unknown>
```

```ts
type Raw = Record<string, unknown>
const str = (v: unknown): string | undefined => (typeof v === 'string' ? v : undefined)

export function normalizeScryfall(json: unknown): CardData {
  const raw = (json ?? {}) as Raw
  const name = str(raw.name) ?? '?'
  const layout = str(raw.layout) ?? 'unknown'
  if (layout !== 'normal') throw new UnsupportedLayout(name, layout)
  const power = str(raw.power), toughness = str(raw.toughness), loyalty = str(raw.loyalty)
  const imageUris = (raw.image_uris ?? {}) as Raw
  const card: CardData = {
    name,
    manaCost: str(raw.mana_cost) ?? '',
    typeLine: str(raw.type_line) ?? '',
    oracleText: str(raw.oracle_text) ?? '',
  }
  if (power !== undefined && toughness !== undefined) card.stat = { kind: 'pt', power, toughness }
  else if (loyalty !== undefined) card.stat = { kind: 'loyalty', value: loyalty }
  const art = str(imageUris.art_crop)
  if (art) card.artUrl = art
  return card
}
```

```ts
export type Token =
  | { kind: 'word'; text: string; glue: boolean }
  | { kind: 'symbol'; code: string; glue: boolean }
  | { kind: 'break' }

const CHUNK = /\{([^}]+)\}|([^{]+)/g

/** Split oracle text into words, {symbols} and paragraph breaks. `glue` = no space before this token. */
export function tokenize(text: string): Token[] {
  const out: Token[] = []
  const paragraphs = text.split('\n')
  paragraphs.forEach((para, pi) => {
    if (pi > 0) out.push({ kind: 'break' })
    for (const chunk of para.split(/\s+/)) {
      if (!chunk) continue
      let first = true
      for (const m of chunk.matchAll(CHUNK)) {
        const glue = !first
        if (m[1] !== undefined) out.push({ kind: 'symbol', code: m[1], glue })
        else out.push({ kind: 'word', text: m[2], glue })
        first = false
      }
    }
  })
  return out
}

/** "+3:", "−2:", "-1:", "0:" — a planeswalker ability prefix, used for hanging indents. */
export function isAbilityPrefix(text: string): boolean {
  return /^[+\-−]?\d+:$/.test(text)
}
```

- [ ] **Step 5: Write parse.ts**

```ts
import { MANA_PATHS } from '../../assets/mana/paths'

export type SymbolSpec =
  | { kind: 'glyph'; glyph: string }
  | { kind: 'hybrid'; a: string; b: string }
  | { kind: 'phyrexian'; color: string }
  | { kind: 'text'; text: string }

const ALIASES: Record<string, string> = { t: 'tap', q: 'untap', '½': 'half', '∞': 'infinity', chaos: 'chaos' }

function glyphFor(part: string): string | undefined {
  const key = ALIASES[part] ?? part
  return key in MANA_PATHS ? key : undefined
}

/** Interpret the text inside a Scryfall {…} symbol. Never throws. */
export function parseSymbol(code: string): SymbolSpec {
  const lower = code.toLowerCase()
  const parts = lower.split('/')
  if (parts.length >= 2 && parts[parts.length - 1] === 'p') {
    const color = glyphFor(parts[0])
    if (color) return { kind: 'phyrexian', color }
  }
  if (parts.length === 2) {
    const a = glyphFor(parts[0]), b = glyphFor(parts[1])
    if (a && b) return { kind: 'hybrid', a, b }
  }
  const g = glyphFor(lower)
  if (g) return { kind: 'glyph', glyph: g }
  return { kind: 'text', text: code }
}
```

```ts
import type { RenderEnv, CanvasLike, Ctx2D } from '../env'
import { MANA_PATHS, MANA_VIEWBOX } from '../../assets/mana/paths'
import { parseSymbol } from './parse'

export interface Stamp { size: number; canvas: CanvasLike }

const OUTLINE = 2

export class StampCache {
  private cache = new Map<string, Stamp>()
  constructor(private env: RenderEnv, private fontFamily: string) {}

  get(code: string, size: number): Stamp {
    const key = `${code}@${size}`
    let s = this.cache.get(key)
    if (!s) { s = this.draw(code, size); this.cache.set(key, s) }
    return s
  }

  private draw(code: string, size: number): Stamp {
    const canvas = this.env.createCanvas(size, size)
    const ctx = canvas.getContext('2d')
    const c = size / 2
    ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, size, size)
    ctx.beginPath(); ctx.arc(c, c, c - OUTLINE / 2 - 0.5, 0, Math.PI * 2)
    ctx.fillStyle = '#fff'; ctx.fill()
    ctx.strokeStyle = '#000'; ctx.lineWidth = OUTLINE; ctx.stroke()
    ctx.fillStyle = '#000'
    const spec = parseSymbol(code)
    switch (spec.kind) {
      case 'glyph': this.glyph(ctx, spec.glyph, c, c, size * 0.62); break
      case 'phyrexian': this.glyph(ctx, 'p', c, c, size * 0.62); break
      case 'hybrid': {
        ctx.beginPath(); ctx.moveTo(size - OUTLINE, OUTLINE); ctx.lineTo(OUTLINE, size - OUTLINE); ctx.stroke()
        this.glyph(ctx, spec.a, size * 0.34, size * 0.34, size * 0.4)
        this.glyph(ctx, spec.b, size * 0.66, size * 0.66, size * 0.4)
        break
      }
      case 'text': this.text(ctx, spec.text, c, c, size); break
    }
    return { size, canvas }
  }

  /** Draw a 32x32 glyph centred at (cx, cy) scaled so the viewbox spans `extent` dots. */
  private glyph(ctx: Ctx2D, name: string, cx: number, cy: number, extent: number) {
    const paths = MANA_PATHS[name]
    if (!paths) return
    const k = extent / MANA_VIEWBOX
    ctx.save(); ctx.translate(cx - extent / 2, cy - extent / 2); ctx.scale(k, k)
    for (const d of paths) ctx.fill(this.env.createPath(d))
    ctx.restore()
  }

  private text(ctx: Ctx2D, text: string, cx: number, cy: number, size: number) {
    let px = Math.round(size * 0.6)
    ctx.textBaseline = 'middle'
    for (; px >= 6; px--) {
      ctx.font = `bold ${px}px ${this.fontFamily}`
      if (ctx.measureText(text).width <= size * 0.7) break
    }
    const w = ctx.measureText(text).width
    ctx.fillText(text, cx - w / 2, cy)
    ctx.textBaseline = 'alphabetic'
  }
}
```

```ts
import type { Ctx2D } from '../env'
import { isAbilityPrefix, type Token } from './tokenize'

export type LineItem =
  | { kind: 'text'; text: string; x: number; width: number }
  | { kind: 'symbol'; code: string; x: number; width: number }
export interface Line { items: LineItem[]; width: number }

export interface WrapOptions {
  ctx: Ctx2D
  font: string
  symbolSize: number
  widthForLine: (lineIndex: number) => number
}

export function wrapText(tokens: Token[], opts: WrapOptions): Line[] {
  const { ctx, symbolSize } = opts
  ctx.font = opts.font
  const space = ctx.measureText(' ').width
  const measure = (t: string) => ctx.measureText(t).width

  const lines: Line[] = []
  let items: LineItem[] = []
  let x = 0
  let indent = 0
  let paragraphStart = true

  const flush = () => {
    const width = items.length ? items[items.length - 1].x + items[items.length - 1].width : 0
    lines.push({ items, width })
    items = []; x = indent
  }

  for (const tok of tokens) {
    if (tok.kind === 'break') { flush(); indent = 0; x = 0; paragraphStart = true; continue }
    const maxW = opts.widthForLine(lines.length)
    const width = tok.kind === 'symbol' ? symbolSize + 2 : measure(tok.text)
    const gap = items.length === 0 || tok.glue ? 0 : space
    if (items.length > 0 && x + gap + width > maxW) { flush() }
    const startX = items.length === 0 ? x : x + gap

    if (tok.kind === 'word' && startX + width > maxW && width > maxW - indent) {
      // Overlong word: emit character runs that fit.
      let rest = tok.text, cx = startX
      while (rest.length) {
        let n = rest.length
        while (n > 1 && cx + measure(rest.slice(0, n)) > opts.widthForLine(lines.length)) n--
        const piece = rest.slice(0, n), pw = measure(piece)
        items.push({ kind: 'text', text: piece, x: cx, width: pw })
        rest = rest.slice(n)
        if (rest.length) { flush(); cx = x }
        else x = cx + pw
      }
    } else if (tok.kind === 'word') {
      items.push({ kind: 'text', text: tok.text, x: startX, width })
      x = startX + width
    } else {
      items.push({ kind: 'symbol', code: tok.code, x: startX, width })
      x = startX + width
    }

    if (paragraphStart) {
      paragraphStart = false
      if (tok.kind === 'word' && isAbilityPrefix(tok.text)) indent = width + space
    }
  }
  if (items.length) flush()
  return lines
}
```

```ts
import type { Ctx2D } from '../env'
import type { StampCache } from '../symbols/stamp'
import type { Line } from './wrap'

/** Paint wrapped lines. `top` is the top of the first line; text baseline sits at 75 % of lineHeight. */
export function drawLines(ctx: Ctx2D, lines: Line[], x: number, top: number, lineHeight: number, font: string, stamps: StampCache, symbolSize: number): void {
  ctx.font = font
  ctx.fillStyle = '#000'
  ctx.textBaseline = 'alphabetic'
  const baseline = Math.round(lineHeight * 0.75)
  lines.forEach((line, i) => {
    const y = top + i * lineHeight
    for (const item of line.items) {
      if (item.kind === 'text') ctx.fillText(item.text, x + item.x, y + baseline)
      else ctx.drawImage(stamps.get(item.code, symbolSize).canvas, x + item.x + 1, y + Math.round((lineHeight - symbolSize) / 2))
    }
  })
}
```

```ts
import type { ImageDataLike } from '../env'
import type { GreyImage } from './types'

export function imageDataToGrey(id: ImageDataLike): GreyImage {
  const n = id.width * id.height, data = new Float32Array(n)
  for (let i = 0; i < n; i++) {
    const r = id.data[i * 4], g = id.data[i * 4 + 1], b = id.data[i * 4 + 2]
    data[i] = (0.299 * r + 0.587 * g + 0.114 * b) / 255
  }
  return { width: id.width, height: id.height, data }
}

/** Stretch so that `cutoffPercent` of pixels clip at each end (like PIL autocontrast). */
export function autocontrast(g: GreyImage, cutoffPercent: number): GreyImage {
  const sorted = Float32Array.from(g.data).sort()
  const n = sorted.length
  if (n === 0) return g
  const k = Math.min(n - 1, Math.floor((n * cutoffPercent) / 100))
  const lo = sorted[k], hi = sorted[n - 1 - k]
  const span = hi - lo
  const data = new Float32Array(n)
  for (let i = 0; i < n; i++) data[i] = span <= 1e-6 ? g.data[i] : Math.min(1, Math.max(0, (g.data[i] - lo) / span))
  return { width: g.width, height: g.height, data }
}

/** gamma > 1 brightens midtones (thermal paper darkens), gamma < 1 darkens. */
export function gammaLift(g: GreyImage, gamma: number): GreyImage {
  const data = new Float32Array(g.data.length)
  const inv = 1 / gamma
  for (let i = 0; i < data.length; i++) data[i] = Math.pow(g.data[i], inv)
  return { width: g.width, height: g.height, data }
}
```

```ts
import type { GreyImage } from './types'

/** Scale so the source covers w×h, centre-crop the excess, then box-filter resample. */
export function resizeCover(src: GreyImage, w: number, h: number): GreyImage {
  const scale = Math.max(w / src.width, h / src.height)
  const cropW = w / scale, cropH = h / scale
  const x0 = (src.width - cropW) / 2, y0 = (src.height - cropH) / 2
  const data = new Float32Array(w * h)
  for (let dy = 0; dy < h; dy++) {
    const sy0 = y0 + (dy * cropH) / h, sy1 = y0 + ((dy + 1) * cropH) / h
    for (let dx = 0; dx < w; dx++) {
      const sx0 = x0 + (dx * cropW) / w, sx1 = x0 + ((dx + 1) * cropW) / w
      let sum = 0, weight = 0
      for (let sy = Math.floor(sy0); sy < Math.ceil(sy1); sy++) {
        const wy = Math.min(sy + 1, sy1) - Math.max(sy, sy0)
        if (wy <= 0) continue
        const row = Math.min(src.height - 1, Math.max(0, sy)) * src.width
        for (let sx = Math.floor(sx0); sx < Math.ceil(sx1); sx++) {
          const wx = Math.min(sx + 1, sx1) - Math.max(sx, sx0)
          if (wx <= 0) continue
          const v = src.data[row + Math.min(src.width - 1, Math.max(0, sx))]
          sum += v * wx * wy; weight += wx * wy
        }
      }
      data[dy * w + dx] = weight > 0 ? sum / weight : 0
    }
  }
  return { width: w, height: h, data }
}
```

```ts
import type { GreyImage, BitImage } from './types'

export function threshold(g: GreyImage): BitImage {
  const bits = new Uint8Array(g.data.length)
  for (let i = 0; i < bits.length; i++) bits[i] = g.data[i] < 0.5 ? 1 : 0
  return { width: g.width, height: g.height, bits }
}

type Kernel = Array<[dx: number, dy: number, weight: number]>
function diffuse(g: GreyImage, kernel: Kernel): BitImage {
  const { width: w, height: h } = g
  const buf = Float32Array.from(g.data)
  const bits = new Uint8Array(w * h)
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
    const i = y * w + x, old = buf[i], nw = old < 0.5 ? 0 : 1
    bits[i] = nw === 0 ? 1 : 0
    const err = old - nw
    for (const [dx, dy, wt] of kernel) {
      const nx = x + dx, ny = y + dy
      if (nx >= 0 && nx < w && ny < h) buf[ny * w + nx] += err * wt
    }
  }
  return { width: w, height: h, bits }
}

/** Atkinson: diffuses 6/8 of the error, which keeps highlights clean. */
export const atkinson = (g: GreyImage) => diffuse(g, [[1, 0, 1 / 8], [2, 0, 1 / 8], [-1, 1, 1 / 8], [0, 1, 1 / 8], [1, 1, 1 / 8], [0, 2, 1 / 8]])
export const floydSteinberg = (g: GreyImage) => diffuse(g, [[1, 0, 7 / 16], [-1, 1, 3 / 16], [0, 1, 5 / 16], [1, 1, 1 / 16]])

const BAYER = [[0, 8, 2, 10], [12, 4, 14, 6], [3, 11, 1, 9], [15, 7, 13, 5]]
export function bayer4(g: GreyImage): BitImage {
  const bits = new Uint8Array(g.data.length)
  for (let y = 0; y < g.height; y++) for (let x = 0; x < g.width; x++) {
    const t = (BAYER[y % 4][x % 4] + 0.5) / 16
    bits[y * g.width + x] = g.data[y * g.width + x] < t ? 1 : 0
  }
  return { width: g.width, height: g.height, bits }
}

export function upscale(b: BitImage, block: number): BitImage {
  if (block === 1) return b
  const w = b.width * block, h = b.height * block, bits = new Uint8Array(w * h)
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) bits[y * w + x] = b.bits[Math.floor(y / block) * b.width + Math.floor(x / block)]
  return { width: w, height: h, bits }
}
```

```ts
import type { Settings } from '../settings'
import { autocontrast, gammaLift } from './grey'
import { resizeCover } from './resize'
import { atkinson, bayer4, floydSteinberg, threshold, upscale } from './dither'
import type { GreyImage, BitImage } from './types'

const DITHERS = { atkinson, 'floyd-steinberg': floydSteinberg, bayer4, threshold } as const

/** Full art pipeline: contrast → gamma → cover-resize to box/block → dither → upscale. Output is exactly box-sized. */
export function prepareArt(src: GreyImage, box: { w: number; h: number }, s: Settings): BitImage {
  const block = s.block
  const small = resizeCover(gammaLift(autocontrast(src, s.contrastCutoff), s.gamma), Math.ceil(box.w / block), Math.ceil(box.h / block))
  const bits = upscale(DITHERS[s.dither](small), block)
  if (bits.width === box.w && bits.height === box.h) return bits
  const out = new Uint8Array(box.w * box.h)
  for (let y = 0; y < box.h; y++) out.set(bits.bits.subarray(y * bits.width, y * bits.width + box.w), y * box.w)
  return { width: box.w, height: box.h, bits: out }
}
```

```ts
import { ART_MAX, ART_MID, ART_MIN, ART_STEP, TEXT_MIN, TEXT_FLOOR } from './constants'

export interface FitCandidate { textSize: number; artH: number; stripped: boolean }

/** The ordered list of (text size, art height, reminder stripped) combinations to try. Spec 5.5. */
export function fitCandidates(baseTextSize: number, stripSetting: boolean): FitCandidate[] {
  const out: FitCandidate[] = []
  const textPhase = (stripped: boolean) => {
    for (let t = baseTextSize; t >= TEXT_MIN; t--) out.push({ textSize: t, artH: ART_MID, stripped })
    for (let a = ART_MID - ART_STEP; a >= ART_MIN; a -= ART_STEP) out.push({ textSize: TEXT_MIN, artH: a, stripped })
  }
  for (let a = ART_MAX; a >= ART_MID; a -= ART_STEP) out.push({ textSize: baseTextSize, artH: a, stripped: stripSetting })
  // the first text-phase entry (base, 180) duplicates the last art entry; skip it
  const before = out.length
  textPhase(stripSetting); out.splice(before, 1)
  if (!stripSetting) textPhase(true)
  for (let t = TEXT_MIN - 1; t >= TEXT_FLOOR; t--) out.push({ textSize: t, artH: ART_MIN, stripped: true })
  return out
}
```

```ts
import type { CardData } from '../card/types'
import type { Settings } from '../settings'
import type { RenderEnv } from '../env'
import { cardHeight } from '../printer'
import { tokenize } from '../text/tokenize'
import { wrapText, type Line } from '../text/wrap'
import { stripReminder } from './reminder'
import { fitCandidates } from './candidates'
import { MARGIN, GAP, NAME_H, TYPE_H, TEXT_PAD, STAT_H, STAT_PAD, COST_SYMBOL, NAME_MIN, LINE_HEIGHT_RATIO } from './constants'

export interface Rect { x: number; y: number; w: number; h: number }
export interface Geometry {
  card: { w: number; h: number }
  nameBar: Rect; nameFontSize: number; costOnSecondLine: boolean; costSymbolSize: number
  art: Rect; typeBar: Rect
  textBox: Rect; textFontSize: number; lineHeight: number; symbolSize: number; textLines: Line[]
  statBox?: Rect
  overflow: boolean; strippedReminder: boolean
}

const symbolCount = (cost: string) => (cost.match(/\{[^}]+\}/g) ?? []).length
export const lineHeightFor = (fontSize: number) => Math.round(fontSize * LINE_HEIGHT_RATIO)
export const symbolSizeFor = (fontSize: number) => Math.round(fontSize * 0.95)

export function fit(card: CardData, s: Settings, env: RenderEnv): Geometry {
  const W = s.dotsPerLine, H = cardHeight(W)
  const ctx = env.createCanvas(8, 8).getContext('2d')
  const family = s.fontFamily
  const innerW = W - 2 * MARGIN

  // --- name bar ---
  const costW = symbolCount(card.manaCost) * (COST_SYMBOL + 3)
  const barInner = innerW - 16
  let costOnSecondLine = costW > barInner / 2
  let nameFontSize = s.nameSize
  const nameRoom = costOnSecondLine ? barInner : barInner - costW - 6
  for (; nameFontSize > NAME_MIN; nameFontSize--) {
    ctx.font = `bold ${nameFontSize}px ${family}`
    if (ctx.measureText(card.name).width <= nameRoom) break
  }
  const nameBar: Rect = { x: MARGIN, y: MARGIN, w: innerW, h: costOnSecondLine ? NAME_H + COST_SYMBOL + 4 : NAME_H }

  // --- stat box width ---
  let statBox: Rect | undefined
  let statW = 0
  if (card.stat) {
    ctx.font = `bold 24px ${family}`
    const label = card.stat.kind === 'pt' ? `${card.stat.power}/${card.stat.toughness}` : card.stat.value
    statW = Math.ceil(ctx.measureText(label).width) + 2 * STAT_PAD
  }

  // --- art + text share the rest ---
  const typeBarH = TYPE_H
  const fixedTop = nameBar.y + nameBar.h + GAP
  const available = H - MARGIN - fixedTop - GAP - typeBarH - GAP   // art + text box
  const textInnerW = innerW - 2 * TEXT_PAD

  let chosen: { lines: Line[]; textSize: number; artH: number; stripped: boolean; overflow: boolean } | undefined
  for (const cand of fitCandidates(s.textSize, s.stripReminder)) {
    const text = cand.stripped ? stripReminder(card.oracleText) : card.oracleText
    const lh = lineHeightFor(cand.textSize)
    const boxH = available - cand.artH - GAP
    const linesAvail = Math.floor((boxH - 2 * TEXT_PAD) / lh)
    // lines whose bottom would overlap the stat box get narrowed
    const widthForLine = (i: number) => {
      if (!card.stat) return textInnerW
      const bottom = TEXT_PAD + (i + 1) * lh
      return bottom > boxH - STAT_H - 2 ? textInnerW - statW - 4 : textInnerW
    }
    const lines = wrapText(tokenize(text), { ctx, font: `${cand.textSize}px ${family}`, symbolSize: symbolSizeFor(cand.textSize), widthForLine })
    if (lines.length <= linesAvail) { chosen = { lines, ...cand, overflow: false }; break }
    chosen = { lines: lines.slice(0, Math.max(1, linesAvail)), ...cand, overflow: true }
  }
  const c = chosen!
  const artH = c.artH
  const art: Rect = { x: MARGIN, y: fixedTop, w: innerW, h: artH }
  const typeBar: Rect = { x: MARGIN, y: art.y + art.h + GAP, w: innerW, h: typeBarH }
  const textBox: Rect = { x: MARGIN, y: typeBar.y + typeBar.h + GAP, w: innerW, h: H - MARGIN - (typeBar.y + typeBar.h + GAP) }
  if (card.stat) statBox = { x: textBox.x + textBox.w - statW, y: textBox.y + textBox.h - STAT_H, w: statW, h: STAT_H }

  return {
    card: { w: W, h: H },
    nameBar, nameFontSize, costOnSecondLine, costSymbolSize: COST_SYMBOL,
    art, typeBar,
    textBox, textFontSize: c.textSize, lineHeight: lineHeightFor(c.textSize), symbolSize: symbolSizeFor(c.textSize), textLines: c.lines,
    statBox,
    overflow: c.overflow, strippedReminder: c.stripped,
  }
}
```

```ts
import type { BitImage } from '../art/types'
import type { CanvasLike, ImageDataLike } from '../env'

/** Pack rows MSB-first, 1 = black. This is the ESC/POS raster layout. */
export function toPacked(b: BitImage): Uint8Array {
  const stride = Math.ceil(b.width / 8)
  const out = new Uint8Array(stride * b.height)
  for (let y = 0; y < b.height; y++) for (let x = 0; x < b.width; x++) {
    if (b.bits[y * b.width + x]) out[y * stride + (x >> 3)] |= 0x80 >> (x & 7)
  }
  return out
}

export function bitImageToImageData(b: BitImage): ImageDataLike {
  const data = new Uint8ClampedArray(b.width * b.height * 4)
  for (let i = 0; i < b.bits.length; i++) {
    const v = b.bits[i] ? 0 : 255
    data[i * 4] = v; data[i * 4 + 1] = v; data[i * 4 + 2] = v; data[i * 4 + 3] = 255
  }
  return { width: b.width, height: b.height, data }
}

/** Threshold a painted greyscale canvas at 50 %. */
export function canvasToBitImage(canvas: CanvasLike): BitImage {
  const id = canvas.getContext('2d').getImageData(0, 0, canvas.width, canvas.height)
  const bits = new Uint8Array(canvas.width * canvas.height)
  for (let i = 0; i < bits.length; i++) {
    const lum = 0.299 * id.data[i * 4] + 0.587 * id.data[i * 4 + 1] + 0.114 * id.data[i * 4 + 2]
    bits[i] = lum < 128 ? 1 : 0
  }
  return { width: canvas.width, height: canvas.height, bits }
}
```

```ts
  ctx.lineWidth = lw; ctx.strokeStyle = '#000'
  ctx.beginPath(); ctx.moveTo(r.x + lw / 2, r.y + lw / 2); ctx.lineTo(r.x + r.w - lw / 2, r.y + lw / 2)
  ctx.lineTo(r.x + r.w - lw / 2, r.y + r.h - lw / 2); ctx.lineTo(r.x + lw / 2, r.y + r.h - lw / 2); ctx.closePath(); ctx.stroke()
}

/** Paint the card (text, bars, symbols) in greyscale, threshold it, then blit the dithered art bits. */
export function paint(card: CardData, g: Geometry, artBits: BitImage | null, s: Settings, env: RenderEnv, stamps: StampCache): BitImage {
  const canvas = env.createCanvas(g.card.w, g.card.h)
  const ctx = canvas.getContext('2d')
  const family = s.fontFamily
  ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, g.card.w, g.card.h)
  strokeRect(ctx, { x: 0, y: 0, w: g.card.w, h: g.card.h }, FRAME)

  // name bar
  strokeRect(ctx, g.nameBar, 2)
  ctx.fillStyle = '#000'; ctx.font = `bold ${g.nameFontSize}px ${family}`; ctx.textBaseline = 'alphabetic'
  ctx.fillText(card.name, g.nameBar.x + 8, g.nameBar.y + Math.round(34 * 0.68))
  const codes = (card.manaCost.match(/\{([^}]+)\}/g) ?? []).map((m) => m.slice(1, -1))
  if (codes.length) {
    const step = g.costSymbolSize + 3
    const y = g.costOnSecondLine ? g.nameBar.y + 34 : g.nameBar.y + Math.round((34 - g.costSymbolSize) / 2)
    let x = g.nameBar.x + g.nameBar.w - 8 - codes.length * step + 3
    for (const code of codes) { ctx.drawImage(stamps.get(code, g.costSymbolSize).canvas, x, y); x += step }
  }

  // art frame (art itself is blitted after thresholding)
  strokeRect(ctx, g.art, 2)

  // type bar
  strokeRect(ctx, g.typeBar, 2)
  ctx.fillStyle = '#000'; ctx.font = `bold 18px ${family}`
  ctx.fillText(card.typeLine, g.typeBar.x + 8, g.typeBar.y + Math.round(g.typeBar.h * 0.7))

  // text box
  strokeRect(ctx, g.textBox, 2)
  drawLines(ctx, g.textLines, g.textBox.x + TEXT_PAD, g.textBox.y + TEXT_PAD, g.lineHeight, `${g.textFontSize}px ${family}`, stamps, g.symbolSize)

  // stat box
  if (g.statBox && card.stat) {
    ctx.fillStyle = '#fff'; ctx.fillRect(g.statBox.x, g.statBox.y, g.statBox.w, g.statBox.h)
    strokeRect(ctx, g.statBox, 2)
    const label = card.stat.kind === 'pt' ? `${card.stat.power}/${card.stat.toughness}` : card.stat.value
    ctx.fillStyle = '#000'; ctx.font = `bold 24px ${family}`

  if (artBits) {
    const ax = g.art.x + 2, ay = g.art.y + 2
    for (let y = 0; y < artBits.height; y++) bits.bits.set(artBits.bits.subarray(y * artBits.width, (y + 1) * artBits.width), (ay + y) * bits.width + ax)
  }
  return bits
}
```

```ts
export function normalizeName(s: string): string {
  return s.normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/æ/gi, 'ae').toLowerCase().trim()
}

function bigrams(s: string): Set<string> {
  const out = new Set<string>()
  for (let i = 0; i < s.length - 1; i++) out.add(s.slice(i, i + 2))
  return out
}
function dice(a: Set<string>, b: Set<string>): number {
  if (a.size === 0 || b.size === 0) return 0
  let hit = 0
  for (const x of a) if (b.has(x)) hit++
  return (2 * hit) / (a.size + b.size)
}

export interface Matcher { search(query: string, limit?: number): string[] }

/** Ranked name lookup: prefix (3) > word prefix (2) > substring (1) > bigram similarity (0..1). */
export function createMatcher(names: string[]): Matcher {
  const entries = names.map((name) => ({ name, norm: normalizeName(name), grams: null as Set<string> | null }))
  return {
    search(query, limit = 8) {
      const q = normalizeName(query)
      if (!q) return []
      const scored: { name: string; score: number }[] = []
      for (const e of entries) {
        let score = 0
        if (e.norm.startsWith(q)) score = 3
        else if (e.norm.includes(' ' + q)) score = 2
        else if (e.norm.includes(q)) score = 1
        if (score) scored.push({ name: e.name, score })
      }
      if (scored.length < limit && q.length >= 4) {
        const qg = bigrams(q)
        for (const e of entries) {
          if (!e.grams) e.grams = bigrams(e.norm)
          const d = dice(qg, e.grams)
          if (d >= 0.5 && !scored.some((s) => s.name === e.name)) scored.push({ name: e.name, score: d })
        }
      }
      scored.sort((a, b) => b.score - a.score || a.name.length - b.name.length || a.name.localeCompare(b.name))
      return scored.slice(0, limit).map((s) => s.name)
    },
  }
}
```

```ts
import { ArtUnavailable, CardNotFound, NetworkError } from '../core/errors'

export interface KeyValue {
  get<T>(key: string): Promise<T | undefined>
  set(key: string, value: unknown): Promise<void>
}

export function memoryKeyValue(): KeyValue {
  const m = new Map<string, unknown>()
  return { async get<T>(k: string) { return m.get(k) as T | undefined }, async set(k, v) { m.set(k, v) } }
}

export interface ScryfallClient {
  fetchCard(exactName: string): Promise<unknown>
  fetchArtBlob(url: string): Promise<Blob>
}

export interface ScryfallOptions {
  fetchFn?: typeof fetch
  cache: KeyValue
  minIntervalMs?: number
  sleep?: (ms: number) => Promise<void>
  now?: () => number
}

const API = 'https://api.scryfall.com'

export function createScryfallClient(o: ScryfallOptions): ScryfallClient {
  const fetchFn = o.fetchFn ?? fetch
  const minInterval = o.minIntervalMs ?? 100
  const sleep = o.sleep ?? ((ms) => new Promise<void>((r) => setTimeout(r, ms)))
  const now = o.now ?? (() => Date.now())
  let last = -Infinity
  let queue: Promise<void> = Promise.resolve()

  /** Serialise requests and keep at least minInterval between them. */
  const throttled = <T>(fn: () => Promise<T>): Promise<T> => {
    const run = queue.then(async () => {
      const wait = last + minInterval - now()
      if (wait > 0) await sleep(wait)
      last = now()
      return fn()
    })
    queue = run.then(() => undefined, () => undefined)
    return run
  }

  const get = async (url: string, init?: RequestInit): Promise<Response> => {
    try { return await fetchFn(url, init) } catch { throw new NetworkError() }
  }

  return {
    async fetchCard(exactName) {
      const key = `card:${exactName}`
      const hit = await o.cache.get<unknown>(key)
      if (hit) return hit
      const url = `${API}/cards/named?exact=${encodeURIComponent(exactName)}`
      const init = { headers: { Accept: 'application/json' } }
      let res = await throttled(() => get(url, init))
      if (res.status === 429) { await sleep(1000); res = await throttled(() => get(url, init)) }
      if (res.status === 404) throw new CardNotFound(exactName)
      if (!res.ok) throw new NetworkError(`Scryfall responded with HTTP ${res.status}.`)
      const card: unknown = await res.json()
      await o.cache.set(key, card)
      return card
    },
    async fetchArtBlob(url) {
      const key = `art:${url}`
      const hit = await o.cache.get<Blob>(key)
      if (hit) return hit
      let res: Response
      try { res = await throttled(() => fetchFn(url)) } catch { throw new ArtUnavailable() }
      if (!res.ok) throw new ArtUnavailable()
      const blob = await res.blob()
      await o.cache.set(key, blob)
      return blob
    },
  }
}
```

```ts
import { useCallback, useRef, useState } from 'react'
import type { Services } from './services'
import type { Settings } from '../core/settings'
import type { CardData } from '../core/card/types'
import type { GreyImage, BitImage } from '../core/art/types'
import type { Geometry } from '../core/layout/fit'
import { normalizeScryfall } from '../core/card/normalize'
import { renderProxy } from '../core/render/renderProxy'
import { ArtUnavailable } from '../core/errors'
import { ensureFonts } from './browserEnv'
import { useEffect } from 'react'

export type ProxyState = 'idle' | 'loading' | 'rendered' | 'error'

export function useProxy(services: Services, settings: Settings) {
  const [state, setState] = useState<ProxyState>('idle')
  const [card, setCard] = useState<CardData | null>(null)
  const [art, setArt] = useState<GreyImage | null>(null)
  const [bits, setBits] = useState<BitImage | null>(null)
  const [geometry, setGeometry] = useState<Geometry | null>(null)
  const [errors, setErrors] = useState<Error[]>([])
  const seq = useRef(0)

  const render = useCallback(async (name: string) => {
    const my = ++seq.current
    setState('loading'); setErrors([])
    try {
      const raw = await services.client.fetchCard(name)
      const c = normalizeScryfall(raw)
      let g: GreyImage | null = null
      const soft: Error[] = []
      if (c.artUrl) {
        try { g = await services.decodeArt(await services.client.fetchArtBlob(c.artUrl)) }
        catch { soft.push(new ArtUnavailable()) }
      }
      if (my !== seq.current) return
      setCard(c); setArt(g); setErrors(soft); setState('rendered')
    } catch (e) {
      if (my !== seq.current) return
      setCard(null); setArt(null); setBits(null); setGeometry(null)
      setErrors([e instanceof Error ? e : new Error(String(e))]); setState('error')
    }
  }, [services])

  // Re-render whenever the card, art or settings change.
  useEffect(() => {
    if (!card) return
    let on = true
    ensureFonts(settings.fontFamily).then(() => {
      if (!on) return
      const r = renderProxy(card, art, settings, services.env)
      setBits(r.bits); setGeometry(r.geometry)
    })
    return () => { on = false }
  }, [card, art, settings, services.env])

  return { state, card, bits, geometry, errors, render }
}
```
